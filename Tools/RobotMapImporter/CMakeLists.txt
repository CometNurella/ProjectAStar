cmake_minimum_required(VERSION 3.20)
project(RobotMapImporter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs highgui)
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(robot_map_importer src/main.cpp src/map.cpp src/ocr.cpp)
target_compile_features(robot_map_importer PRIVATE cxx_std_17)
target_include_directories(robot_map_importer PRIVATE include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(robot_map_importer PRIVATE ${OpenCV_LIBS} nlohmann_json::nlohmann_json)
if(WIN32)
  target_sources(robot_map_importer PRIVATE src/windows_ocr.cpp)
  target_compile_definitions(robot_map_importer PRIVATE MAPIMPORT_WINDOWS_OCR NOMINMAX)
  target_link_libraries(robot_map_importer PRIVATE windowsapp)
endif()
if(MSVC)
  target_compile_options(robot_map_importer PRIVATE /W4 /utf-8 /permissive-)
else()
  target_compile_options(robot_map_importer PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(map_geometry_tests tests/geometry_tests.cpp src/map.cpp)
  target_compile_features(map_geometry_tests PRIVATE cxx_std_17)
  target_include_directories(map_geometry_tests PRIVATE include ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(map_geometry_tests PRIVATE ${OpenCV_LIBS} nlohmann_json::nlohmann_json)
  add_test(NAME map_geometry COMMAND map_geometry_tests)
  add_executable(ocr_parser_tests tests/ocr_tests.cpp src/ocr.cpp)
  target_compile_features(ocr_parser_tests PRIVATE cxx_std_17)
  target_include_directories(ocr_parser_tests PRIVATE include ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(ocr_parser_tests PRIVATE ${OpenCV_LIBS})
  add_test(NAME ocr_parser COMMAND ocr_parser_tests)
endif()
